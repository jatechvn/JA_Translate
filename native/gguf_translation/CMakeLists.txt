cmake_minimum_required(VERSION 3.15)
project(ja_gguf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(LIB llama ggml)
  add_library(${LIB} SHARED IMPORTED)
  set_target_properties(${LIB} PROPERTIES IMPORTED_IMPLIB "${ROOT}/.local_ai_tools/gguf-import/${LIB}.lib")
endforeach()
add_library(ja_gguf SHARED bridge.cpp)
target_include_directories(ja_gguf PRIVATE "${ROOT}/.local_ai_tools/llama.cpp/include" "${ROOT}/.local_ai_tools/llama.cpp/ggml/include")
target_compile_options(ja_gguf PRIVATE /utf-8 /EHsc)
target_link_libraries(ja_gguf PRIVATE llama ggml)
